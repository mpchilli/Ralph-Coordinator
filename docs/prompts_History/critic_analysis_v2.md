```
import os
import google.generativeai as genai
from colorama import Fore, Style

# Load Config
# (In production, load from config.yaml, here we default for simplicity)
MODELS = {
    "planner": "gemini-2.0-pro-exp-02-05",
    "executor": "gemini-2.5-flash-preview-09-25",
    "vision": "gemini-2.0-flash-exp"
}

def configure():
    api_key = os.environ.get("GEMINI_API_KEY")
    if not api_key:
        raise ValueError("GEMINI_API_KEY not found in environment")
    genai.configure(api_key=api_key)

def generate(prompt, model_role="executor"):
    """Generates text using the routed model."""
    configure()
    model_name = MODELS.get(model_role, MODELS["executor"])
    model = genai.GenerativeModel(model_name)
  
    try:
        response = model.generate_content(prompt)
        return response.text
    except Exception as e:
        return f"LLM ERROR: {e}"

def generate_vision(prompt, image_path):
    """Generates reasoning based on an image."""
    configure()
    if not os.path.exists(image_path):
        return "ERROR: Image not found"
      
    model = genai.GenerativeModel(MODELS["vision"])
  
    # Upload/Load image logic (simplified for local file)
    with open(image_path, "rb") as f:
        img_data = f.read()
  
    response = model.generate_content([
        {'mime_type': 'image/png', 'data': img_data},
        prompt
    ])
    return response.text
```

```
import os
import subprocess
import shlex

class Referee:
    """Enforces security and updates visibility (Captain's Badge)."""
  
    def __init__(self, allowlist):
        self.allowlist = allowlist

    def update_badge(self, task_id, status, log_entry):
        """Writes the .ralph-status.md Heads-Up Display."""
        badge_path = os.path.join(os.getcwd(), ".ralph-status.md")
        icon = "🟢" if status == "ACTIVE" else "🔴" if status == "ERROR" else "⚪"
      
        content = f"""# {icon} Ralph Status: {status}
**Current Task:** `{task_id}`
**Last Action:** {log_entry}

---
*Open this file in VS Code Preview (Ctrl+Shift+V) to monitor.*
"""
        with open(badge_path, "w", encoding="utf-8") as f:
            f.write(content)

    def validate_command(self, cmd_list):
        if not cmd_list: raise ValueError("Empty command")
        binary = cmd_list[0]
        if binary not in self.allowlist:
            if not any(binary.endswith(a) for a in self.allowlist):
                raise ValueError(f"Command '{binary}' blocked by Referee.")

    def run_safe(self, cmd_str):
        cmd_list = shlex.split(cmd_str, posix=False)
        self.validate_command(cmd_list)
        # Check=True raises CalledProcessError on non-zero exit
        subprocess.run(cmd_str, shell=True, check=True)
```

```
import os
import time
from .safety import Referee
from .state import PlanManager
from .llm import generate, generate_vision
# Import Playwright sync API
from playwright.sync_api import sync_playwright

class MicroRatchet:
    def __init__(self):
        self.referee = Referee(["git", "npm", "ls", "echo", "python", "node", "npx"])
        self.state = PlanManager()

    def run_cycle(self, task_id):
        # 1. Lock & Badge
        lock_file = 'coordinator.lock'
        if os.path.exists(lock_file): return
        with open(lock_file, 'w') as f: f.write("LOCKED")
      
        self.referee.update_badge(task_id, "ACTIVE", "Starting Cycle")

        try:
            # 2. Context
            plan = self.state.load_plan()
          
            # 3. Detect Hat (UI vs Code)
            is_ui_task = "[UI]" in task_id
          
            # 4. Generate Code
            prompt = f"""
            ROLE: {'UI Designer' if is_ui_task else 'Senior Developer'}
            TASK: {task_id}
            CONTEXT: {plan}
          
            Write the code to solve this task. 
            Output ONLY the code blocks (filename and content).
            """
            response = generate(prompt, "executor" if not is_ui_task else "vision")
            print(f"Generated Code for {task_id}")
            self.referee.update_badge(task_id, "ACTIVE", "Code Generated")
          
            # (In production, we need a parser here to actually write the files from 'response')
            # For now, we assume the Agent is running a command to write files.

            # 5. Verify
            try:
                if is_ui_task:
                    self._verify_ui(task_id)
                else:
                    self.referee.run_safe("npm test")
              
                # 6. Commit (Green)
                self.referee.run_safe(f"git commit -am 'feat: {task_id}'")
                self.referee.update_badge(task_id, "SUCCESS", "Committed")
              
            except Exception as e:
                # 7. Reset (Red)
                print(f"Verification Failed: {e}")
                self.referee.run_safe("git reset --hard")
                self.referee.update_badge(task_id, "ERROR", f"Reverted: {str(e)}")

        finally:
            if os.path.exists(lock_file): os.remove(lock_file)

    def _verify_ui(self, task_id):
        """Designer Hat: Take screenshot and verify."""
        with sync_playwright() as p:
            browser = p.chromium.launch()
            page = browser.new_page()
            page.goto("http://localhost:3000") # Assume dev server running
            page.screenshot(path="snapshot.png")
            browser.close()
          
        # Check with Vision Model
        verdict = generate_vision(f"Does this UI satisfy: {task_id}? Reply MATCH or MISMATCH.", "snapshot.png")
        if "MISMATCH" in verdict:
            raise ValueError(f"Visual Regression: {verdict}")
```
